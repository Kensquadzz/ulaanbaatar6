cmake_minimum_required(VERSION 3.16)
project(ub6core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT MSVC)
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_library(ub6core STATIC src/Geo.cpp src/Solar.cpp src/LiveClock.cpp src/Weather.cpp
            src/Streaming.cpp src/Quality.cpp src/RoadGraph.cpp src/Gameplay.cpp src/Mesh.cpp src/Vehicle.cpp src/TouchControls.cpp)
target_include_directories(ub6core PUBLIC include)
add_executable(ub6tool tools/ub6tool.cpp)
add_executable(meshcheck tools/meshcheck.cpp)
target_link_libraries(meshcheck ub6core)
target_link_libraries(ub6tool ub6core)
enable_testing()
add_executable(ub6tests tests/tests.cpp)
target_link_libraries(ub6tests ub6core)
target_compile_definitions(ub6tests PRIVATE UB6_DATA_DIR="${CMAKE_SOURCE_DIR}/../data/tiles")
add_test(NAME ub6tests COMMAND ub6tests)
